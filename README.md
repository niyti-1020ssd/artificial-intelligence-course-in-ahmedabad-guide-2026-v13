```python
# Example of a simple linear regression model in Python
import numpy as np
from sklearn.linear_model import LinearRegression

# Sample data
X = np.array([[1], [2], [3], [4], [5]])
y = np.array([2, 3, 5, 7, 11])

# Create and train the model

# Predictions

print(predictions)
```
